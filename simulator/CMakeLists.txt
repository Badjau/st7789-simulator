cmake_minimum_required(VERSION 3.20)
project(dashboard_sim C)
if(NOT LVGL_DIR)
  message(FATAL_ERROR "Set LVGL_DIR to the pinned LVGL checkout")
endif()
file(TO_CMAKE_PATH "${LVGL_DIR}" LVGL_DIR)
file(GLOB_RECURSE LVGL_SOURCES "${LVGL_DIR}/src/*.c")
add_executable(dashboard_sim main.c ../src/dashboard.c ${LVGL_SOURCES})
target_include_directories(dashboard_sim PRIVATE . ../include "${LVGL_DIR}" "${LVGL_DIR}/..")
target_compile_definitions(dashboard_sim PRIVATE LV_CONF_INCLUDE_SIMPLE)
